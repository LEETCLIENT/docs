# Авто прыжок
```lua
module.set("Auto Jump", MOVEMENT)

local mc = internal.call("net.minecraft.client.MinecraftClient", "getInstance")
events.set_callback(event_tick, function(data)
    local player = internal.get_field(mc, "player")
    local grounded = internal.get_field(player, "onGround")
    if grounded then
        internal.call(player, "jump")
    end
end)
```
