# Проверка лицензии

```lua
module.set("example http", COMBAT)

local hasAccess = false

local function onResponse(response)
    if response.status == 200 then
        for uid in string.gmatch(response.body, "[^\r\n]+") do
            if uid == tostring(client.get_uid()) then
                client.chat("Лицензия подтверждена.")
                hasAccess = true
                break
            end
        end
    end
    if not hasAccess then
        client.chat("У вас нет доступа к скрипту")
    end
end

function enabled()
    network.get("https://pastebin.com/raw/73EwTxNH", {}, onResponse)
end

events.set_callback(event_tick, function(data)
    if not hasAccess then return end
    -- логика
end)
```
